+        /// 直接从 ICC 字节写 iCCP（2026-11-15）。用于标准 ICC 已在内存中生成、但落临时文件
+        /// 被 Windows ACL/Defender 拒绝的场景：不落盘 ICC，直接把字节压进 PNG。
+        /// </summary>
+        public static bool TryWriteIccpBytes(string pngPath, byte[] icc, string profileName = "icc")
+        {
+            if (icc.Length < 132) return false;
+            for (int attempt = 0; ; attempt++)
+            {
+                try
+                {
+                    var name = string.IsNullOrWhiteSpace(profileName) ? "icc" : profileName;
+                    if (name.Length > 79) name = name[..79];
+                    using var compressed = new MemoryStream();
+                    using (var z = new ZLibStream(compressed, CompressionLevel.Optimal, leaveOpen: true))
+                        z.Write(icc, 0, icc.Length);
+                    var payload = new MemoryStream();
+                    payload.Write(Encoding.ASCII.GetBytes(name));
+                    payload.WriteByte(0);
+                    payload.WriteByte(0); // compression method: zlib
+                    compressed.Position = 0;
+                    compressed.CopyTo(payload);
+                    var chunk = BuildChunk("iCCP", payload.ToArray());
+
+                    var data = File.ReadAllBytes(pngPath);
+                    if (!TryParseHeader(data, out _, out _, out var ihdrEnd)) return false;
+
+                    // 移除旧 iCCP，再在 IHDR 后插入（必须在 IDAT 前）。
+                    using var ms = new MemoryStream(data.Length + chunk.Length + 16);
+                    ms.Write(data, 0, ihdrEnd);
+                    int pos = ihdrEnd;
+                    while (pos + 12 <= data.Length)
+                    {
+                        int len = BinaryPrimitives.ReadInt32BigEndian(data.AsSpan(pos));
+                        var type = Encoding.ASCII.GetString(data, pos + 4, 4);
+                        if (type == "iCCP") { pos += 12 + len; continue; }
+                        ms.Write(data, pos, 12 + len);
+                        pos += 12 + len;
+                    }
+                    var buf = ms.ToArray();
+                    using var outMs = new MemoryStream(buf.Length + chunk.Length);
+                    outMs.Write(buf, 0, ihdrEnd);
+                    outMs.Write(chunk);
+                    outMs.Write(buf, ihdrEnd, buf.Length - ihdrEnd);
+                    File.WriteAllBytes(pngPath, outMs.ToArray());
+                    return true;
+                }
+                catch (IOException) when (attempt < 2)
+                {
+                    // exiftool/pngopt 退出瞬间可能仍持有输出句柄；共享冲突是暂时的，退避后重试。
+                    Thread.Sleep(100 * (attempt + 1));
+                }
+                catch
+                {
+                    return false;
+                }
+            }
+        }

+        /// </summary>

+        {
+            removed = 0;
+            try
+            {
+                var data = File.ReadAllBytes(pngPath);

+                    if (type == "cICP") { hasCicp = true; break; }

+                ms.Write(data, 0, ihdrEnd);
+                for (int pos = ihdrEnd; pos + 12 <= data.Length;)

+                    if (type == "iCCP") { removed++; pos += 12 + len; continue; }
+                    ms.Write(data, pos, 12 + len);
+                    pos += 12 + len;
+                }
+                if (removed == 0) return true;
+                File.WriteAllBytes(pngPath, ms.ToArray());

+            }

+        /// 读回 PNG/APNG 的 <c>iCCP</c> 并解压出原始 ICC 字节。用于源 ICC 的纯托管提取，
+        /// 避免 exiftool 在目标文件上留下 `_exiftool_tmp` 后反而读不到自己刚写出的 profile。
+        /// 无 iCCP、压缩方法非 0、zlib 解压失败或损坏 → null（不猜）。
+        /// </summary>
+        public static byte[]? TryReadIccp(string pngPath)
+        {
+            try
+            {
+                var data = File.ReadAllBytes(pngPath);
+                if (!TryParseHeader(data, out _, out _, out var ihdrEnd)) return null;
+                int pos = ihdrEnd;
+                while (pos + 12 <= data.Length)
+                {
+                    int len = BinaryPrimitives.ReadInt32BigEndian(data.AsSpan(pos));
+                    if (len < 0 || pos + 12 + len > data.Length) return null;
+                    var type = Encoding.ASCII.GetString(data, pos + 4, 4);
+                    if (type == "iCCP")
+                    {
+                        var payload = data.AsSpan(pos + 8, len);

+                        using var z = new ZLibStream(input, CompressionMode.Decompress);
+                        using var output = new MemoryStream();
+                        z.CopyTo(output);
+                        return output.ToArray();
+                    }
+                    pos += 12 + len;
+                    if (type == "IEND") break;
+                }
+                return null;
+            }
